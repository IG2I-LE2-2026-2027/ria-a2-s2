Focus sur les fermetures (aka "closure")

http://www.w3schools.com/js/js_function_closures.asp 
var compteur=0; 

function trace(s) {
	document.write(compteur + ": " + s + "<br />");
	compteur++;
} 

function mkTrace() {
	var compteur_interne=0;
	return function(s){
		document.write(compteur_interne + " " + s + "<br />");
		compteur_interne++;
	}
}
  • Chaque appel à la fonction mkTrace génère un scope qui ne disparaît pas après la fin de l'exécution de la fonction.
  • Chacune des fonctions trace manipule son propre scope qui contient une variable compteur_interne différente.
var trace2 = mkTrace(); 
var trace3 = mkTrace(); 

trace2("appel 1 (trace 2)");
trace2("appel 2 (trace 2)");
trace3("appel 1 (trace 3)");
trace3("appel 2 (trace 3)");
trace3("appel 3 (trace 3)");
trace2("appel 3 (trace 2)");
  • On peur renvoyer un objet contenant plusieurs méthodes ayant un scope commun 

function mkTrace2() {
	var compteur_interne=0;
	return {
		trace : function(s){
			document.write(compteur_interne + ": " + s + "<br />");
			compteur_interne++;
		}, 
		reset : function(){compteur_interne=0;},
		getCompteur : function(){return compteur_interne;},
		setCompteur : function(value){compteur_interne = value;},
	}; 
}

var oTrace = mkTrace2(); 
oTrace.trace("appel 1");
oTrace.trace("appel 2");
document.write("Après deux appels, compteur vaut " + oTrace.getCompteur() + "<br />"); 
oTrace.reset();
document.write("Après reset, compteur vaut " + oTrace.getCompteur() + "<br />");
oTrace.trace("appel 3");
oTrace.setCompteur(5);
document.write("Après setcompteur, compteur vaut " + oTrace.getCompteur() + "<br />");
oTrace.trace("appel 4");
